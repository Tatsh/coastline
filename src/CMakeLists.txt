# Each subsystem directory builds its static library and lists its sources explicitly. A
# recursive glob would hide what is built. A source added to the tree would join the build with no
# record of the decision, and a source removed would disappear the same way.
#
# Static libraries are deliberate. Archiving produces object files and archives without
# linking a ROM. The reconstruction is partial, and no link can succeed until every routine a call
# arrives at has a body.

# Everything the subsystem libraries share. An interface library provides the settings without
# compiling anything itself.
add_library(coastline_common INTERFACE)
# A reconstructed routine retains the parameters of the original, used or not.
target_compile_options(coastline_common INTERFACE -Wall -Wextra -Wno-unused-parameter)
target_include_directories(coastline_common INTERFACE "${CMAKE_SOURCE_DIR}/src"
                                                    "${CMAKE_SOURCE_DIR}/src/libultra/include")

# Subsystems join this list as their directories are reconstructed.
set(COASTLINE_SUBSYSTEMS)

foreach(SUBSYSTEM ${COASTLINE_SUBSYSTEMS})
  add_subdirectory(${SUBSYSTEM})
endforeach()

# The entry points sit outside the subsystems. The Windows port is C++, and this directory links its
# libraries, so the language is enabled here. The Nintendo 64 entry is assembly.
if(TGR_PLATFORM STREQUAL "win")
  enable_language(CXX)
  add_subdirectory(platform/win)
else()
  add_library(coastline_main OBJECT main.s)
  target_link_libraries(coastline_main PRIVATE coastline_common)
endif()

# Every archive the build produces, for a target that builds all of them at once.
set(COASTLINE_LIBRARIES coastline_main)
foreach(SUBSYSTEM ${COASTLINE_SUBSYSTEMS})
  list(APPEND COASTLINE_LIBRARIES coastline_${SUBSYSTEM})
endforeach()

if(TGR_PLATFORM STREQUAL "n64")
  string(TOUPPER "${TGR_REGION}" _tgr_region_upper)
  # Assembly generated by splat provides the routines and data of the original image not yet
  # converted to C. A release without the assembly builds its code alone.
  set(_asm_list "${CMAKE_SOURCE_DIR}/asm/${TGR_REGION}/sources.cmake")
  if(EXISTS "${_asm_list}")
    include("${_asm_list}")
    list(TRANSFORM COASTLINE_ASM_${_tgr_region_upper} PREPEND "${CMAKE_SOURCE_DIR}/")
    add_library(coastline_asm OBJECT ${COASTLINE_ASM_${_tgr_region_upper}})
    target_include_directories(coastline_asm PRIVATE "${CMAKE_SOURCE_DIR}/asm")
    target_compile_options(coastline_asm PRIVATE "-Wa,-I${CMAKE_SOURCE_DIR}/asm")
    list(APPEND COASTLINE_LIBRARIES coastline_asm)
    set(COASTLINE_ASM_SYMBOLS "${CMAKE_SOURCE_DIR}/asm/${TGR_REGION}/undefined_syms.ld"
                              "${CMAKE_SOURCE_DIR}/asm/${TGR_REGION}/undefined_funcs.ld")
  endif()

  # The game links the inflate side of the vendored zlib 1.0.4. The warnings concern upstream
  # style.
  set(_zlib "${CMAKE_SOURCE_DIR}/3rdparty/zlib-1.0.4")
  add_library(
    coastline_zlib STATIC
    "${_zlib}/adler32.c" "${_zlib}/infblock.c" "${_zlib}/infcodes.c" "${_zlib}/inffast.c"
    "${_zlib}/inflate.c" "${_zlib}/inftrees.c" "${_zlib}/infutil.c" "${_zlib}/uncompr.c"
    "${_zlib}/zutil.c")
  target_include_directories(coastline_zlib PUBLIC "${_zlib}")
  # The game's changes to zlib use the game's state and libultra.
  target_include_directories(coastline_zlib PRIVATE "${CMAKE_SOURCE_DIR}/src"
                                                    "${CMAKE_SOURCE_DIR}/src/libultra/include")
  target_compile_options(coastline_zlib PRIVATE -w)
  target_link_libraries(coastline_common INTERFACE coastline_zlib)
  list(APPEND COASTLINE_LIBRARIES coastline_zlib)

  add_subdirectory(libultra)
  list(APPEND COASTLINE_LIBRARIES ${COASTLINE_LIBULTRA_LIBRARIES})

  # The game code of the Nintendo 64 releases. A routine joins this list when its body moves from
  # the assembly of the original into C.
  set(_game_sources audio.c audiomix.s camera.c carselect.c controllerscreen.c debugtext.c decaldraw.c decaleditor.c driving.c effects.c fade.c game.c hud.c input.c heap.c mainmenu.c mathutil.c model.c options.c paintshop.c pakmenu.c physics.c playerselect.c race.c

  list(APPEND _game_sources bodyforces.c camerachase.c cameracollision.c displaylistops.c
                           frameclear.c playerinput.c profileinit.c renderinit.c screen.c seasontrack.c

  # These sources write the Japanese text in UTF-8. The Japanese release compiles copies with the

  if(TGR_REGION STREQUAL "jp")
    foreach(_source carselect.c decaleditor.c options.c pakmenu.c paintshop.c playerselect.c splash.c

  add_library(coastline_game STATIC ${_game_sources})
  target_link_libraries(coastline_game PRIVATE coastline_common)
  list(APPEND COASTLINE_LIBRARIES coastline_game)

add_custom_target(archives DEPENDS ${COASTLINE_LIBRARIES})

# A Windows library cannot link until every routine has a body, and TGR_LINK turns its link on. A
# release whose routines all have a body, in C or in the assembly of the original, links whenever
# the assets of its cartridge are extracted. Without extracted assets, the archives are the check.
if(TGR_PLATFORM STREQUAL "win")
  if(TGR_LINK)
    if(TGR_RENDERER STREQUAL "d3d")
      set(_library BRD3D)
    else()
      set(_library BRGlide)
    endif()
    add_library(${_library} SHARED $<TARGET_OBJECTS:coastline_main>)
    set_target_properties(${_library} PROPERTIES PREFIX "")
    target_link_libraries(${_library} PRIVATE ${COASTLINE_LIBRARIES} ole32 user32)
  endif()
else()
  if(COASTLINE_ASSETS_DIR AND COASTLINE_ASM_SYMBOLS)
    # The program is flattened into the bytes the IPL3 loads. build-rom puts the bytes after the
    # original header and IPL3, places the assets at their offsets, and writes the checksum.
    set(_link_script "${CMAKE_CURRENT_SOURCE_DIR}/platform/n64/coastline-${TGR_REGION}.ld")
    # The archives come after the objects, in one group. Their routines call each other. An
    # archive member is not linked when an object already defines its routine (the assembly of a
    # vendored zlib routine, for one). libgcc provides the helpers GCC calls for the arithmetic the
    # processor does not do in 32-bit mode.
    set(_link_libraries ${COASTLINE_LIBRARIES})
    list(REMOVE_ITEM _link_libraries coastline_main coastline_asm)
    add_executable(${CMAKE_PROJECT_NAME} $<TARGET_OBJECTS:coastline_main>)
    target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE coastline_asm "-Wl,--start-group"
                                                        ${_link_libraries} "-Wl,--end-group" gcc)
    set(_link_scripts "${_link_script}" ${COASTLINE_ASM_SYMBOLS})
    list(TRANSFORM _link_scripts PREPEND "-T")
    target_link_options(${CMAKE_PROJECT_NAME} PRIVATE -nostdlib -nostartfiles ${_link_scripts})
    set_property(TARGET ${CMAKE_PROJECT_NAME} APPEND PROPERTY LINK_DEPENDS "${_link_script}"
                                                                ${COASTLINE_ASM_SYMBOLS})
    set(_program "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}.bin")
    set(_rom "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}-${TGR_REGION}.z64")
    add_custom_command(
      OUTPUT "${_program}"
      COMMAND "${CMAKE_OBJCOPY}" -O binary $<TARGET_FILE:${CMAKE_PROJECT_NAME}> "${_program}"
      DEPENDS ${CMAKE_PROJECT_NAME}
      COMMENT "Flattening the program"
      VERBATIM)
    add_custom_command(
      OUTPUT "${_rom}"
      COMMAND "${COASTLINE_TOOL_BUILD_ROM}" "${COASTLINE_ASSETS_DIR}/boot.bin" "${_program}"
              "${COASTLINE_ASSETS_DIR}" "${_rom}"
      DEPENDS "${_program}" "${COASTLINE_ASSETS_DIR}/manifest.txt" "${COASTLINE_TOOL_BUILD_ROM}"
              coastline_tools
      COMMENT "Building ${_rom}"
      VERBATIM)
    add_custom_target(rom ALL DEPENDS "${_rom}")
  endif()
endif()
