# Each source directory builds its static library from its own CMakeLists.txt and lists its sources
# explicitly. A recursive glob would hide what is built. A source added to the tree would join the
# build with no record of the decision, and a source removed would disappear the same way.
#
# Static libraries are deliberate. Archiving produces object files and archives without linking a
# ROM. The reconstruction is partial, and no link can succeed until every routine a call arrives at
# has a body.

# The settings every library of the game shares. An interface library provides them without
# compiling anything itself.
if(TGR_PLATFORM STREQUAL "n64" AND NOT TGR_IDO)
  # Keep the ROM's size optimisation after CMake's build-type flags.
  add_compile_options($<$<COMPILE_LANGUAGE:C>:-Os>)
endif()
add_library(coastline_options INTERFACE)
# Reconstructed routines retain unused parameters and narrow intermediates to original field widths.
# IDO takes the flags of tools/ido_flags.txt alone.
if(MSVC)
  target_compile_options(coastline_options INTERFACE /W3 /wd4100 /wd4200 /wd4244 /wd5030)
  target_compile_definitions(coastline_options INTERFACE _CRT_SECURE_NO_WARNINGS)
elseif(NOT TGR_IDO)
  target_compile_options(coastline_options INTERFACE -Wall -Wextra -Wno-unused-parameter)
endif()
target_include_directories(coastline_options INTERFACE "${CMAKE_SOURCE_DIR}/src"
                                                     "${CMAKE_SOURCE_DIR}/src/common/libultra/include")

if(TGR_PLATFORM STREQUAL "win")
  if(MSVC)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
  endif()
  if(MSVC AND DXSDK_DIR)
    set(_dxsdk_headers "${CMAKE_BINARY_DIR}/dxsdk-headers")
    file(MAKE_DIRECTORY "${_dxsdk_headers}")
    file(COPY_FILE "${DXSDK_DIR}/Include/dplay.h" "${_dxsdk_headers}/dplay.h")
    file(COPY_FILE "${DXSDK_DIR}/Include/dplobby.h" "${_dxsdk_headers}/dplobby.h")
    include_directories(BEFORE "${_dxsdk_headers}")
    link_directories("${DXSDK_DIR}/Lib/x86")
  endif()
  target_include_directories(coastline_options INTERFACE "${CMAKE_SOURCE_DIR}/src/common")
  # The Windows port is C++, and this directory links its libraries, so the language is enabled
  # here.
  if(NOT MSVC)
    target_compile_options(coastline_options INTERFACE "$<$<COMPILE_LANGUAGE:CXX>:-x;c++>")
  endif()
  enable_language(CXX)
  if(TGR_SHARED_WIN_BUILD)
    if(MSVC)
      set(_shared_archive_suffix .lib)
      set(_shared_archive_prefix "")
      set(_shared_archive_config "Release/")
    else()
      set(_shared_archive_suffix .a)
      set(_shared_archive_prefix lib)
      set(_shared_archive_config "")
    endif()
    set(_shared_archive_dir "${TGR_SHARED_WIN_BUILD}/src")
    add_library(coastline_ear STATIC IMPORTED GLOBAL)
    string(CONCAT _shared_ear_archive "${_shared_archive_dir}/common/ear/"
           "${_shared_archive_config}${_shared_archive_prefix}coastline_ear"
           "${_shared_archive_suffix}")
    set_target_properties(
      coastline_ear PROPERTIES IMPORTED_LOCATION "${_shared_ear_archive}")
    add_library(coastline_win32_ear STATIC IMPORTED GLOBAL)
    string(CONCAT _shared_win32_ear_archive "${_shared_archive_dir}/win32/ear/"
           "${_shared_archive_config}${_shared_archive_prefix}coastline_win32_ear"
           "${_shared_archive_suffix}")
    set_target_properties(
      coastline_win32_ear PROPERTIES IMPORTED_LOCATION "${_shared_win32_ear_archive}")
    target_link_libraries(coastline_ear INTERFACE coastline_win32_ear)
    target_link_libraries(coastline_win32_ear INTERFACE coastline_ear)
  else()
    add_subdirectory(common/ear)
  endif()
  add_subdirectory(win32)
else()
  # The Nintendo 64 releases find their headers in n64 as well as here.
  target_include_directories(coastline_options INTERFACE "${CMAKE_SOURCE_DIR}/src/n64")
  add_subdirectory(n64)
  add_subdirectory(n64/libultra)
endif()
add_subdirectory(common)
if(TGR_PLATFORM STREQUAL "n64")
  include(n64/libultra/common-sources.cmake)

# Every archive the build produces, for a target that builds all of them at once.
set(COASTLINE_LIBRARIES coastline_main coastline_common ${COASTLINE_COMMON_LIBULTRA_LIBRARIES}
                        ${COASTLINE_LIBULTRA_LIBRARIES})
if(TGR_PLATFORM STREQUAL "win")
  list(APPEND COASTLINE_LIBRARIES coastline_win32)
  if(TGR_RENDERER STREQUAL "d3d")
    list(APPEND COASTLINE_LIBRARIES coastline_win32_d3d)
  else()
    list(APPEND COASTLINE_LIBRARIES coastline_win32_glide)

  list(APPEND COASTLINE_LIBRARIES coastline_n64)

if(TGR_PLATFORM STREQUAL "n64")
  string(TOUPPER "${TGR_REGION}" _tgr_region_upper)
  # Assembly generated by splat provides the routines and data of the original image not yet
  # converted to C. A release without the assembly builds its code alone.
  set(_asm_list "${CMAKE_SOURCE_DIR}/src/n64/asm/${TGR_REGION}/sources.cmake")
  if(EXISTS "${_asm_list}")
    include("${_asm_list}")
    list(TRANSFORM COASTLINE_ASM_${_tgr_region_upper} PREPEND "${CMAKE_SOURCE_DIR}/")
    add_library(coastline_asm OBJECT ${COASTLINE_ASM_${_tgr_region_upper}})
    target_include_directories(coastline_asm PRIVATE "${CMAKE_SOURCE_DIR}/src/n64/asm")
    target_compile_options(coastline_asm PRIVATE "-Wa,-I${CMAKE_SOURCE_DIR}/src/n64/asm")
    list(APPEND COASTLINE_LIBRARIES coastline_asm)
    set(COASTLINE_ASM_SYMBOLS
        "${CMAKE_SOURCE_DIR}/src/n64/asm/${TGR_REGION}/undefined_syms.ld"
        "${CMAKE_SOURCE_DIR}/src/n64/asm/${TGR_REGION}/undefined_funcs.ld")
  endif()

  # The game links the inflate side of the vendored zlib 1.0.4. The warnings concern upstream
  # style.
  set(_zlib "${CMAKE_SOURCE_DIR}/3rdparty/zlib-1.0.4")
  add_library(
    coastline_zlib STATIC
    "${_zlib}/adler32.c" "${_zlib}/infblock.c" "${_zlib}/infcodes.c" "${_zlib}/inffast.c"
    "${_zlib}/inflate.c" "${_zlib}/inftrees.c" "${_zlib}/infutil.c" "${_zlib}/uncompr.c"
    "${_zlib}/zutil.c")
  target_include_directories(coastline_zlib PUBLIC "${_zlib}")
  # The game's changes to zlib use the game's state and libultra.
  target_include_directories(coastline_zlib PRIVATE "${CMAKE_SOURCE_DIR}/src"
                                                    "${CMAKE_SOURCE_DIR}/src/common/libultra/include")
  target_compile_options(coastline_zlib PRIVATE -w)
else()
  # The PC libraries link the checksum of the vendored zlib 1.0.4 and not its inflate side.
  set(_zlib "${CMAKE_SOURCE_DIR}/3rdparty/zlib-1.0.4")
  if(TGR_SHARED_WIN_BUILD)
    add_library(coastline_zlib STATIC IMPORTED GLOBAL)
    string(CONCAT _shared_zlib_archive "${_shared_archive_dir}/"
           "${_shared_archive_config}${_shared_archive_prefix}coastline_zlib"
           "${_shared_archive_suffix}")
    set_target_properties(
      coastline_zlib PROPERTIES IMPORTED_LOCATION "${_shared_zlib_archive}")
  else()
    add_library(coastline_zlib STATIC "${_zlib}/adler32.c")
  endif()
  # zlib 1.0.4 defines its routines in the style C23 removed.
  if(NOT TGR_SHARED_WIN_BUILD)
    set_target_properties(coastline_zlib PROPERTIES C_STANDARD 90)
  endif()
  if(TGR_SHARED_WIN_BUILD)
    target_include_directories(coastline_zlib INTERFACE "${_zlib}")
  else()
    target_include_directories(coastline_zlib PUBLIC "${_zlib}")
  endif()
  if(NOT TGR_SHARED_WIN_BUILD)
    if(MSVC)
      target_compile_options(coastline_zlib PRIVATE /W0)
    else()
      target_compile_options(coastline_zlib PRIVATE -w)
    endif()
  endif()
endif()
target_link_libraries(coastline_options INTERFACE coastline_zlib)

if(TGR_PLATFORM STREQUAL "n64" AND TGR_IDO AND CMAKE_BUILD_TYPE STREQUAL "Release" AND NOT
                                                       TGR_ENABLE_PATCHES)
  set(COASTLINE_REFERENCE_ROM
      ""
      CACHE FILEPATH "Original regional ROM to supply the assets and header to the matching build.")
  set(_rom_arguments)
  if(COASTLINE_REFERENCE_ROM)
    list(APPEND _rom_arguments --reference "${COASTLINE_REFERENCE_ROM}")
    set(_rom "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}-${TGR_REGION}.z64")
  else()
    set(_rom "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}-${TGR_REGION}-code-only.z64")
  endif()
  add_custom_target(
    rom
    COMMAND
      "${COASTLINE_TOOL_ASSEMBLE_CODE}" --source-root "${CMAKE_SOURCE_DIR}" --region
      "${TGR_REGION}" --build "${CMAKE_BINARY_DIR}" ${_rom_arguments} --output "${_rom}"
    DEPENDS archives coastline_tools
    COMMENT "Assembling the matching ${_tgr_region_upper} ROM"
    VERBATIM)

if(TGR_PLATFORM STREQUAL "win")
  if(TGR_RENDERER STREQUAL "d3d")
    set(_library BRD3D)
  else()
    set(_library BRGlide)
  endif()
  set(_link_libraries ${COASTLINE_LIBRARIES})
  list(REMOVE_ITEM _link_libraries coastline_main)
  # The Windows libraries contain every routine of their units, called or not, as the original
  # libraries do.
  list(TRANSFORM _link_libraries REPLACE "^(coastline_win32.*)$" "$<LINK_LIBRARY:WHOLE_ARCHIVE,\\1>")
  add_library(${_library} SHARED $<TARGET_OBJECTS:coastline_main>)
  if(TARGET earpds)
    add_dependencies(${_library} earpds)
  endif()
  set_target_properties(${_library} PROPERTIES PREFIX "")
  # The libraries call each other, so they are linked as one group.
  if(MSVC)
    target_link_libraries(${_library} PRIVATE ${_link_libraries} ole32 user32 ddraw dinput dplayx
                                              dxguid winmm msacm32)
  else()
    target_link_libraries(${_library} PRIVATE "-Wl,--start-group" ${_link_libraries}
                                              "-Wl,--end-group" ole32 user32 ddraw dinput dplayx
                                              dxguid winmm msacm32)
  endif()
  if(TGR_RENDERER STREQUAL "glide")
    target_link_libraries(${_library} PRIVATE coastline_glide)
  endif()
else()
  if(NOT TGR_IDO AND COASTLINE_ASSETS_DIR AND COASTLINE_ASM_SYMBOLS)
    # The program is flattened into the bytes the IPL3 loads. build-rom puts the bytes after the
    # original header and IPL3, places the assets at their offsets, and writes the checksum.
    set(_link_script "${CMAKE_CURRENT_SOURCE_DIR}/n64/coastline-${TGR_REGION}.ld")
    # The archives come after the objects, in one group. Their routines call each other. An
    # archive member is not linked when an object already defines its routine (the assembly of a
    # vendored zlib routine, for one). libgcc provides the helpers GCC calls for the arithmetic the
    # processor does not do in 32-bit mode.
    set(_link_libraries ${COASTLINE_LIBRARIES})
    list(REMOVE_ITEM _link_libraries coastline_main coastline_asm)
    add_executable(${CMAKE_PROJECT_NAME} $<TARGET_OBJECTS:coastline_main>)
    target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE coastline_asm "-Wl,--start-group"
                                                        ${_link_libraries} "-Wl,--end-group" gcc)
    set(_link_scripts "${_link_script}" ${COASTLINE_ASM_SYMBOLS})
    list(TRANSFORM _link_scripts PREPEND "-T")
    target_link_options(${CMAKE_PROJECT_NAME} PRIVATE -nostdlib -nostartfiles ${_link_scripts})
    set_property(TARGET ${CMAKE_PROJECT_NAME} APPEND PROPERTY LINK_DEPENDS "${_link_script}"
                                                                ${COASTLINE_ASM_SYMBOLS})
    set(_program "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}.bin")
    set(_rom "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}-${TGR_REGION}.z64")
    add_custom_command(
      OUTPUT "${_program}"
      COMMAND "${CMAKE_OBJCOPY}" -O binary $<TARGET_FILE:${CMAKE_PROJECT_NAME}> "${_program}"
      DEPENDS ${CMAKE_PROJECT_NAME}
      COMMENT "Flattening the program"
      VERBATIM)
    add_custom_command(
      OUTPUT "${_rom}"
      COMMAND "${COASTLINE_TOOL_BUILD_ROM}" "${COASTLINE_ASSETS_DIR}/boot.bin" "${_program}"
              "${COASTLINE_ASSETS_DIR}" "${_rom}"
      DEPENDS "${_program}" "${COASTLINE_ASSETS_DIR}/manifest.txt" "${COASTLINE_TOOL_BUILD_ROM}"
              coastline_tools
      COMMENT "Building ${_rom}"
      VERBATIM)
    add_custom_target(rom ALL DEPENDS "${_rom}")
  endif()
endif()
